sync task, as zpool set feature@physical_rewrite=enabled), the all-redirect run did not, so the two movers started from
different feature states.  1 November: on every fresh copy of an input, before either mover, physical_rewrite is enabled
(with extensible_dataset, its dependency), for the all-redirect mover too.  The hybrid adapter then finds it enabled and
enables nothing (physical_rewrite_enabled_by_adapter=0 in all 48 hybrid runs; physical_rewrite_enabled_before=1 in all 96).
Everything else is as on 29 October: same trees and binaries, same build flags, same niceness, same 8 cases, both movers,
3 runs each per tree, the same correctness checks.

== what differed from the 29 October run
  - the feature state before the move (the change above);
  - inputs: on 29 October each run created its own input with the generator; now each tree's 8 inputs were created once and
    every run used a fresh sparse copy of one (as asked).  The copies' labels name the original member paths; the adapter,
    the helper and the generator all import by scanning the copy's directory, which found the copy in every run;
  - counts: identical in every case, mover and tree for the 3-run value sets of entries, entries after drop-retained,
    cold-import entries, rewritten blocks, redirected DVAs, census round 1, census rewrite/redirect split, the inventory,
    txgs and fill gang blocks;
  - kept bytes: 3 cells whose 3 values moved within the two values both dates show (R 16K 0 % redirect, D 16K 0 % redirect:
    9928704 in all three, against 9920512 once or twice on 29 October; D 16K 100 % redirect: 10002432 in all three, against
    9994240 once): 2 sectors of 4 KiB, a placement difference, not a different set of entries;
  - fill: D 128K 10 % hybrid wrote 4904 new 128K blocks in one of its three fills (4896 in the others and on 29 October);
  - table KiB on disk (log + block-pointer ZAP, as allocated) moves between runs in steps of 28.5-29 KiB on both dates; the
    medians differ in some cells by one step (e.g. R 128K 0 % redirect 259.0 against 230.0), within the ranges both dates show;
  - costs: Nov/Oct ratio of the medians over the 32 tree x case x mover cells: mover s median 1.016 (0.896-1.184), scrub s
    1.073 (1.009-1.167), census walk s 1.010 (0.847-1.150), written MiB 1.005 (0.992-1.013), peak heap 1.003 (0.867-1.226).
    Scrubs were about 7 % slower in every cell, both movers; nothing else moved beyond the run-to-run spread.  Peak heap
    includes libzpool's ARC and dbufs, so it follows what the caches hold, not the table alone.

== what could not be done
  - zpool get all feature@ (needs the kernel module): replaced by feature_tool get, same states;
  - git push (no credentials on this VM): the commit is local, its patch is in the reply;
  - R is still the 29 October release build without zdb (zdb does not compile without ZFS_DEBUG);
  - as on 29 October, the raw member hashes of each run copy around its read-only verify were not taken (the 16 originals
    were hashed before and after all runs, unchanged); the fill ran after drop-retained, never with the snapshot held; and
    the pending cases of the protocol (clones, dedup and BRT sharing, keyless encryption with an ACK-bearing ZIL, gang and
    multi-DVA, crash cuts, too-full pools, readers held across reuse, offline child / return / resilver / replacement,
    kernel) are still pending.

== feature_tool.c (the helper, not committed)
// SPDX-License-Identifier: CDDL-1.0
/*
 * Helper of the 1 November fixture re-run (not committed): the pool
 * features of a generator input, in userspace, where "zpool get" (which
 * needs the kernel module) cannot run.
 *
 *   feature_tool <dir> get              read-only import: one line per
 *                                       feature, as "zpool get all
 *                                       feature@" prints its state
 *   feature_tool <dir> enable <name>    writable import: enable it (and
 *                                       its dependencies), then export
 *
 * The pool "jointfixture" is found by scanning <dir>; no cache file is
 * written outside <dir>.
 */
#include <stdio.h>
#include <string.h>
#include <sys/zfs_context.h>
#include <sys/spa.h>
#include <sys/spa_impl.h>
#include <sys/dmu_tx.h>
#include <sys/dsl_synctask.h>
#include <sys/zfeature.h>
#include <sys/fs/zfs.h>
#include <zfeature_common.h>
#include <libzutil.h>
#include <libzpool.h>

#define	POOL	"jointfixture"

static const char *dir;

static spa_t *
import(boolean_t readonly)
{
	importargs_t ia;
	char *paths[1];
	nvlist_t *config = NULL, *props = NULL;
	spa_t *spa;
	libpc_handle_t lpch = {
		.lpc_lib_handle = NULL,
		.lpc_ops = &libzpool_config_ops,
		.lpc_printerr = B_TRUE
	};

	memset(&ia, 0, sizeof (ia));
	paths[0] = (char *)dir;
	ia.path = paths;
	ia.paths = 1;
	ia.can_be_active = B_TRUE;
	VERIFY0(zpool_find_config(&lpch, POOL, &config, &ia));
	if (!readonly) {
		props = fnvlist_alloc();
		fnvlist_add_string(props, "cachefile", "none");
	}
	VERIFY0(spa_import((char *)POOL, config, props, readonly ?
	    ZFS_IMPORT_SKIP_MMP : ZFS_IMPORT_NORMAL));
	fnvlist_free(config);
	if (props != NULL)
		fnvlist_free(props);
	VERIFY0(spa_open(POOL, &spa, FTAG));
	return (spa);
}

static int
enable_check(void *arg, dmu_tx_t *tx)
{
	spa_feature_t *fid = arg;

	if (spa_feature_is_enabled(dmu_tx_pool(tx)->dp_spa, *fid))
		return (SET_ERROR(EEXIST));
	return (0);
}

static void
enable_sync(void *arg, dmu_tx_t *tx)
{
	spa_feature_t *fid = arg;

	spa_feature_enable(dmu_tx_pool(tx)->dp_spa, *fid, tx);
}

int
main(int argc, char **argv)
{
	char cache[4096];
	spa_t *spa;
	int err;

	if (argc < 3 || argv[1][0] != '/' || (strcmp(argv[2], "get") != 0 &&
	    (strcmp(argv[2], "enable") != 0 || argc != 4))) {
		(void) fprintf(stderr, "usage: feature_tool <dir> get | "
		    "feature_tool <dir> enable <name>\n");
		return (2);
	}
	dir = argv[1];
	(void) snprintf(cache, sizeof (cache), "%s/zpool.cache", dir);
	spa_config_path = cache;
	if (strcmp(argv[2], "get") == 0) {
		kernel_init(SPA_MODE_READ);
		spa = import(B_TRUE);
		for (int i = 0; i < SPA_FEATURES; i++) {
			const char *state = "disabled";

			if (spa_feature_is_active(spa, i))
				state = "active";
			else if (spa_feature_is_enabled(spa, i))
				state = "enabled";
			(void) printf("%s feature@%s %s\n", POOL,
			    spa_feature_table[i].fi_uname, state);
		}
		spa_close(spa, FTAG);
		kernel_fini();
		return (0);
	} else {
		spa_feature_t fid;

		/* kernel_init() fills the feature table. */
		kernel_init(SPA_MODE_READ | SPA_MODE_WRITE);
		if (zfeature_lookup_name(argv[3], &fid) != 0) {
			(void) fprintf(stderr, "unknown feature %s\n", argv[3]);
			kernel_fini();
			return (2);
		}
		spa = import(B_FALSE);
		err = dsl_sync_task(POOL, enable_check, enable_sync, &fid, 0,
		    ZFS_SPACE_CHECK_NONE);
		(void) printf("enable feature@%s: %s\n", argv[3], err == 0 ?
		    "enabled now" : err == EEXIST ? "already enabled" :
		    strerror(err));
		spa_close(spa, FTAG);
		VERIFY0(spa_export(POOL, NULL, B_FALSE, B_FALSE));
		kernel_fini();
		return (err == 0 || err == EEXIST ? 0 : 1);
	}
}

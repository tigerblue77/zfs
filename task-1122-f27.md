**The PR** (#19807, "Keep file deletions working on a full pool", open, head `c2c1e14c8e9`, on upstream master `c913da86156`):
1. `678ff11191c` "Keep the root quota extension away from the next space level" (`module/zfs/dsl_dir.c` only, 14 lines). At the root, `dsl_dir_tempreserve_impl()` lets estimated usage run `quota >> 5` past the quota; commit 1 caps that extension at half the way to the next space level: (RESERVED − NORMAL) / 2 = S/4 for writes, (EXTRA_RESERVED − NETFREE) / 2 = S/16 for netfree transactions (S = the slop space). Its own figures, libzpool, a 4 GiB file pool at `spa_asize_inflation=1`: writes until ENOSPC went 118.9 MiB past the quota (92.9 % of the 128 MiB slop) and a netfree transaction then failed with ENOSPC; with the commit 30.9 MiB and the netfree transaction succeeded. Never measured in the kernel, nor at the default inflation.
2. `e619b675a92` "Let file deletions finish on a full pool": the follow-up transactions of rm, truncate and hole punching marked netfree.
3. `c2c1e14c8e9` "Keep the space that frees need out of reach of admin operations": a NETFREE level (leaves 3/8 of S) between RESERVED (1/2) and EXTRA_RESERVED (1/4); adds the ZTS test `no_space/enospc_free`.

**Why `enospc_free` alone cannot answer it** (computed from master's `vdev_metaslab_set_size()`, `spa_get_slop_space()` and `dsl_dir_tempreserve_impl()`, not run): its pool is one 512 MiB file, that is 31 metaslabs of 16 MiB, 496 MiB, S = 128 MiB (the minimum), NORMAL quota Q = 368 MiB. - **R1, the audit's question:** `enospc_free` PASSES in every B run (at least 3) and every A run (at least 2).

- **SPLIT** if no KEEP reason, R1, R2 for both rows, R3, A clean at 24 (every operation rc 0 on A in p24 and s24, its largest O below S/4), and every measured fill ended with ENOSPC.
- **INCONCLUSIVE** otherwise, with the reasons (an A failure at 24, R3 not met, a fill that timed out or ended otherwise, too few passes).

                break

                break

    rw_k1=$(txquota)

guard() {
    # A measured pass starts only if it should end by GO + 90 min.
    [ -f $L/go.txt ] || return 0
    case $1 in P*) ;; *) return 0 ;; esac
    gd_go=$(cat $L/go.txt)
    gd_dmax=$(sed -n 's/.* pass P[0-9]* done in \([0-9]*\) s$/\1/p' $L/driver.log | sort -n | tail -1)
    gd_now=$(date +%s)
    if [ $(( gd_now + ${gd_dmax:-0} )) -gt $(( gd_go + 5400 )) ]; then
        log "pass $1 NOT RUN: it would end after GO + 90 min (now $gd_now, longest pass ${gd_dmax:-0} s, GO $gd_go)"
        return 5
    fi
    return 0
}

cmd_pass() {
    ps_name=$1
    ps_arms=$2
    ps_rows=$3
    guard $ps_name || exit 5
    log "pass $ps_name start: arms $ps_arms, rows $ps_rows"
    ps_p0=$(date +%s)
    for ps_a in $ps_arms; do
        cmd_load $ps_a
        for ps_r in $ps_rows; do
            cmd_row $ps_name $ps_a $ps_r
        done
    done
    log "pass $ps_name done in $(( $(date +%s) - ps_p0 )) s"
}

cmd_go() {
    [ -f $L/go.txt ] && die "go: $L/go.txt exists already"
    date +%s > $L/go.txt
    log "GO at $(date -u +%FT%TZ)"
}

cmd_quiet() {
    qt_i=0
    while [ $qt_i -lt 30 ]; do
        qt_l1=$(cut -d' ' -f1 /proc/loadavg)
        awk -v l="$qt_l1" 'BEGIN {exit !(l < 1.0)}' && break
        sleep 10
        qt_i=$(( qt_i + 1 ))
    done
    qt_s0=$(head -1 /proc/stat)
    sleep 60
    qt_s1=$(head -1 /proc/stat)
    qt_out=$(printf '%s\n%s\n' "$qt_s0" "$qt_s1" | awk '
        NR == 1 {for (i = 2; i <= 9; i++) a[i] = $i}
        NR == 2 {t = 0; for (i = 2; i <= 9; i++) {d[i] = $i - a[i]; t += d[i]}
            printf "steal %.2f %% idle %.2f %% iowait %.2f %%", 100 * d[9] / t, 100 * d[5] / t, 100 * d[6] / t}')
    qt_line="QUIET $qt_out load1 $(cut -d' ' -f1 /proc/loadavg) (waited $(( qt_i * 10 )) s for load1 < 1.0)"
    echo "$(date -u +%FT%TZ) $qt_line" >> $L/quiet-check.txt
    log "$qt_line"
}

order() {
    # odd passes A then B, even passes B then A
    if [ $(( $1 % 2 )) = 1 ]; then echo "A B"; else echo "B A"; fi
}

cmd_timerule() {
    [ -f $L/go.txt ] || die "timerule: no GO recorded"
    tr_go=$(cat $L/go.txt)
    tr_d1=$(sed -n 's/.* pass P1 done in \([0-9]*\) s$/\1/p' $L/driver.log | tail -1)
    tr_d2=$(sed -n 's/.* pass P2 done in \([0-9]*\) s$/\1/p' $L/driver.log | tail -1)
    [ -n "$tr_d1" ] && [ -n "$tr_d2" ] || die "timerule: P1 and P2 must both be done"
    tr_d=$(( (tr_d1 + tr_d2) / 2 ))
    tr_r=$(( tr_d / 8 ))
    tr_now=$(date +%s)
    tr_lim=$(( tr_go + 5280 ))
    tr_last=10
    tr_drop=""
    tr_proj=$(( tr_now + (tr_last - 2) * tr_d + 60 ))
    while [ $tr_proj -gt $tr_lim ] && [ $tr_last -gt 6 ]; do
        tr_last=$(( tr_last - 1 ))
        tr_proj=$(( tr_now + (tr_last - 2) * tr_d + 60 ))
    done
    if [ $tr_proj -gt $tr_lim ]; then
        tr_drop="P5 P6"
        tr_proj=$(( tr_proj - 4 * tr_r ))
    fi
    if [ $tr_proj -gt $tr_lim ]; then
        tr_last=4
        tr_drop=""
        tr_proj=$(( tr_now + 2 * tr_d + 60 ))
    fi
    : > $T/plan.txt
    tr_k=3
    while [ $tr_k -le $tr_last ]; do
        tr_rows="p24 s24 z24 p1"
        case " $tr_drop " in *" P$tr_k "*) tr_rows="p24 s24 z24" ;; esac
        echo "P$tr_k|$(order $tr_k)|$tr_rows" >> $T/plan.txt
        tr_k=$(( tr_k + 1 ))
    done
    {
        echo "GO $tr_go, now $tr_now, P1 $tr_d1 s, P2 $tr_d2 s, d $tr_d s, row about $tr_r s, limit GO + 88 min = $tr_lim"
        echo "plan: passes P1 to P$tr_last${tr_drop:+, row p1 dropped from $tr_drop}; projected end $tr_proj ($(( (tr_proj - tr_go) / 60 )) min after GO)"
    } > $L/time-rule.txt
    cp $T/plan.txt $L/plan.txt
    log "TIMERULE: $(tail -1 $L/time-rule.txt)"
    cat $T/plan.txt
}

    done 3< $T/plan.txt
    log "REST DONE: $rs_ran planned pass(es) run, $rs_notrun not run (GO + 90 min)"
}

def parse(path, tag):
    out = []

            for kv in w[1:]:

    v = d.get(k, '')

def fmt(x, scale=1):
